# Build libhv inside SereneDB's compiler and libc++ environment.
function(add_web3_libhv)
  set(BUILD_STATIC ON)
  set(BUILD_SHARED OFF)
  set(BUILD_EXAMPLES OFF)
  set(BUILD_UNITTEST OFF)
  set(WITH_OPENSSL OFF)
  set(WITH_CURL OFF)
  set(CMAKE_SKIP_INSTALL_RULES TRUE)
  add_subdirectory("${WEB3_LIBHV_SOURCE_DIR}" libhv EXCLUDE_FROM_ALL SYSTEM)
  set_property(TARGET hv_static PROPERTY CXX_STANDARD 17)
  target_compile_definitions(hv_static PUBLIC WITHOUT_HTTP_CONTENT)
endfunction()
add_web3_libhv()

target_include_directories(web3_thread_pool PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")

target_include_directories(web3_http PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
